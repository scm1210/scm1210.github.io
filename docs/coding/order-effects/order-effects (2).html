function hist(key, { log = false, lo = 0, hi = 10, money = false, truth = null, labels }) {
  const ok = data.filter(d => isFinite(d[key]) && (!log || d[key] > 0))
    .map(d => ({ x: log ? Math.log10(d[key]) : d[key], order: d.g === "A" ? labels[0] : labels[1], raw: d[key] }));
  let thresholds, xopt;
  if (log) {
    const xs = ok.map(d => d.x);
    let a = Math.max(0, Math.floor(Math.min(lo, ...xs) * 2) / 2), b = Math.min(13, Math.ceil(Math.max(hi, ...xs) * 2) / 2);
    thresholds = d3.range(a, b + 0.001, 0.25);
    xopt = { domain: [a, b], ticks: d3.range(Math.ceil(a), Math.floor(b) + 1), tickFormat: v => (money ? "$" : "") + big(10 ** v), label: "log scale" };
  } else {
    thresholds = d3.range(0.5, 11, 1);
    xopt = { domain: [0.5, 10.5], ticks: d3.range(1, 11), label: null };
  }
  // ggplot-style: geom_histogram(position = "identity", alpha = .6, colour = "grey20")
  const bar = { thresholds, stroke: "#333", strokeWidth: 0.6, inset: 0 };
  const marks = split
    ? ["A", "B"].map((g, i) => Plot.rectY(ok.filter(d => d.order === labels[i]),
        Plot.binX({ y: "count" }, { x: "x", fill: GROUP[g], fillOpacity: 0.6, ...bar })))
    : [Plot.rectY(ok, Plot.binX({ y: "count" }, { x: "x", fill: GROUP.pool, fillOpacity: 0.8, ...bar }))];
  marks.push(Plot.ruleY([0]));
  if (truth) marks.push(
    Plot.ruleX([Math.log10(truth)], { stroke: "currentColor", strokeDasharray: "4,3" }),
    Plot.text([Math.log10(truth)], { x: d => d, frameAnchor: "top", dx: 6, textAnchor: "start", text: () => "actual ≈ " + (truth / 1e6).toFixed(1) + "M" }));
  const f = v => !isFinite(v) ? "–" : log ? (money ? "$" : "") + big(v) : v.toFixed(1);
  const cap = split
    ? labels.map(l => `${l}: median ${f(d3.median(ok.filter(d => d.order === l), d => d.raw))}`).join(" · ")
    : `n = ${ok.length} · median ${f(d3.median(ok, d => d.raw))}`;
  return html`${Plot.plot({
    height: 260, marginLeft: 20, x: xopt, y: { label: null, ticks: [] }, marks
  })}${split ? html`<div style="display:flex;gap:18px;font-size:14px;margin:2px 0 4px">
      <span><span style="display:inline-block;width:12px;height:12px;background:${GROUP.A};opacity:.75;border:1px solid #333"></span> Group A · ${labels[0]}</span>
      <span><span style="display:inline-block;width:12px;height:12px;background:${GROUP.B};opacity:.75;border:1px solid #333"></span> Group B · ${labels[1]}</span></div>` : ""}<p style="font-family:monospace;font-size:13px;margin:0">${cap}</p>`;
}

function num(s) {
  if (s == null) return NaN;
  s = String(s).toLowerCase().replace(/[$€£,\s~≈]/g, "").replace(/^about|^around/, "");
  const m = s.match(/^(-?\d*\.?\d+)(.*)$/); if (!m) return NaN;
  let v = parseFloat(m[1]); const suf = m[2];
  if (/^(b|bn|bil)/.test(suf)) v *= 1e9; else if (/^(m|mm|mil)/.test(suf)) v *= 1e6; else if (/^(k|thou)/.test(suf)) v *= 1e3;
  return v;
}

// Columns are matched by keywords in the question text, so both form sections
// can reuse the same wording. Duplicate headers are fine (parsed as raw rows).
function toRows(table) {
  const h = table[0].map(x => x.trim());
  const cols = re => h.map((t, i) => re.test(t) ? i : -1).filter(i => i >= 0);
  const C = { fin: cols(/finland/i), mov: cols(/movie|film/i), hap: cols(/happy/i), sun: cols(/sunny|weather/i) };
  const cond = cols(/odd|even|birth|condition|order|version|group/i)[0];
  const pc = cols(/precept|which section/i).filter(i => i !== cond)[0];
  return table.slice(1).map(r => {
    const pick = k => { for (let j = 0; j < C[k].length; j++) { const v = (r[C[k][j]] || "").trim(); if (v !== "") return { v, j }; } return null; };
    const f = pick("fin"), m = pick("mov"), hh = pick("hap"), s = pick("sun");
    let g = null;
    if (cond != null) { const t = (r[cond] || "").toLowerCase();
      if (/odd|^a\b|finland first/.test(t)) g = "A"; else if (/even|^b\b|movie first/.test(t)) g = "B"; }
    if (!g && f && C.fin.length > 1) g = f.j === 0 ? "A" : "B";
    if (!g && f && m) g = C.fin[f.j] < C.mov[m.j] ? "A" : "B";
    const pm = pc != null ? (r[pc] || "").match(/P\s?0?(\d{1,2})/i) : null;
    return { p: pm ? "P" + pm[1].padStart(2, "0") : "", g: g || "A",
             fin: f ? num(f.v) : NaN, mov: m ? num(m.v) : NaN, hap: hh ? num(hh.v) : NaN, sun: s ? num(s.v) : NaN };
  });
}

function big(v) { if (!isFinite(v)) return "–"; const a = Math.abs(v);
  if (a >= 1e9) return +(v / 1e9).toPrecision(3) + "B"; if (a >= 1e6) return +(v / 1e6).toPrecision(3) + "M";
  if (a >= 1e3) return +(v / 1e3).toPrecision(3) + "K"; return +v.toPrecision(3) + ""; }
function corr(x, y) { const n = x.length; if (n < 3) return NaN; const mx = d3.mean(x), my = d3.mean(y);
  let a = 0, b = 0, c = 0; for (let i = 0; i < n; i++) { a += (x[i] - mx) * (y[i] - my); b += (x[i] - mx) ** 2; c += (y[i] - my) ** 2; }
  return b && c ? a / Math.sqrt(b * c) : NaN; }
fmtR = r => isFinite(r) ? r.toFixed(2) : "–"
rand = i => { const x = Math.sin(i * 12.9898) * 43758.5453; return x - Math.floor(x); }

function demoRows() {
  const n = d3.randomNormal(0, 1), out = [];
  for (let i = 0; i < 36; i++) {
    const g = i % 2 ? "B" : "A", sun = Math.max(1, Math.min(10, Math.round(4 + n() * 2.2)));
    out.push({ p: PRECEPTS[i % PRECEPTS.length].code, g,
      fin: 10 ** ((g === "A" ? 6.75 : 7.55) + n() * 0.35), mov: 10 ** ((g === "A" ? 7.0 : 7.65) + n() * 0.35),
      sun, hap: Math.max(1, Math.min(10, Math.round((g === "A" ? 3 + 0.5 * sun : 6) + n() * 1.4))) });
  }
  return out;
}
